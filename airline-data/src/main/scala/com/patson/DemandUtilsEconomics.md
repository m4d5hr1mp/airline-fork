  )

  // Final indicators with derived fields
  case class CountryEconomicIndicators(
    gdpNominal: Int,            // GDP (Nominal USD, per capita)
    incomePpp: Int,             // GNI-PPP (PPP Adjusted income, per capita)
    incomeDiscretionary: Int,   // Discretionary income (per capita, in PPP-Dollars; derived)
    leisureBudget: Int,         // Leisure spending (average, based on discretionary income; derived)
    gini: Double,               // Income GINI Index (coefficient between 0 and 1)
    pppFactor: Double,          // PPP Adjustment Factor
    exchangeRate: Double        // Exchange rate from local currency to USD
  )

  private val GINI_FALLBACK = 0.4        
  private val ESSENTIALS_FRACTION_BASE = 0.75 // Base fraction of PPP income for essentials (adjusted by Gini)

  // Constants for Leisure Budget Formula. In PPP Dollars!
  private val LEISURE_FRACTION_LOW = 0.025
  private val LEISURE_FRACTION_HIGH = 0.1
  private val LEISURE_LOW_THRESHOLD = 10000
  private val LEISURE_HIGH_THRESHOLD = 75000

  // Compute discretionary income using fast lognormal approximation:
  private def computeDiscretionaryIncome(pppIncome: Int, gini: Double): Int = {
    val effectiveGini = if (gini == 0.0) GINI_FALLBACK else gini
    val essentialsFraction = ESSENTIALS_FRACTION_BASE * (1 + effectiveGini / 2) // Gini increases essentials burden on average household
    val discretionary = math.max(0.0, pppIncome * (1 - math.min(1.0, essentialsFraction))).toInt
    discretionary
  }

  // Compute leisure budget:
  private def computeLeisureBudget(discretionaryIncome: Int): Int = {
    val fraction = if (discretionaryIncome <= LEISURE_LOW_THRESHOLD) {
      LEISURE_FRACTION_LOW
    } else if (discretionaryIncome >= LEISURE_HIGH_THRESHOLD) {
      LEISURE_FRACTION_HIGH
    } else {
      LEISURE_FRACTION_LOW + (LEISURE_FRACTION_HIGH - LEISURE_FRACTION_LOW) * 
        ((discretionaryIncome - LEISURE_LOW_THRESHOLD).toDouble / (LEISURE_HIGH_THRESHOLD - LEISURE_LOW_THRESHOLD))
    }
    (discretionaryIncome * fraction).toInt
  }

  lazy val countryEconomicIndicators: Map[String, CountryEconomicIndicators] = hardCountryEconomicIndicators.map {
    case (code, hard) =>
      val discretionary = computeDiscretionaryIncome(hard.incomePpp, hard.gini)
      val leisure = computeLeisureBudget(discretionary)
      code.toUpperCase -> CountryEconomicIndicators(
        gdpNominal = hard.gdpNominal,
        incomePpp = hard.incomePpp,
        incomeDiscretionary = discretionary,
        leisureBudget = leisure,
        gini = hard.gini,
        pppFactor = hard.pppFactor,
        exchangeRate = hard.exchangeRate
      )
  }

  // Accessor method to retrieve indicators for a country
  def getCountryEconomicIndicators(countryCode: String): Option[CountryEconomicIndicators] = {
    countryEconomicIndicators.get(countryCode.toUpperCase)
  }
